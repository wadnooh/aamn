⚡ الطريقة الوحيدة المضمونة:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  📌 في Visual Studio (المفتوح أمامك الآن):

  1️⃣  انظر في أعلى النافذة
  
  2️⃣  ستجد زر أخضر ▶️ مكتوب بجانبه "SudanTravelApp.API"
  
  3️⃣  اضغط على الزر (أو اضغط F5 على الكيبورد)
  
  4️⃣  انتظر 5-10 ثواني
  
  5️⃣  ستظهر نافذة سوداء (Console) - اتركها مفتوحة!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
